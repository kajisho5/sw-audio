# SW AUDIO — build for CLAP / VST3 / AUv2 (macOS) via clap-wrapper (CLAP-first)
cmake_minimum_required(VERSION 3.21)
if (NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS version")
endif()
project(sw-audio VERSION 0.13.0 LANGUAGES C CXX)
if (APPLE)
    enable_language(OBJC OBJCXX)  # clap-wrapper (AUv2 / VST3 on macOS) compiles Objective-C++ sources
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if (MINGW)
    set(CMAKE_SHARED_MODULE_PREFIX "")  # VST3/CLAP binaries must not get a "lib" prefix
endif()
if (MSVC)
    add_compile_options(/utf-8)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")  # self-contained plugins
endif()

option(SW_BUILD_TESTS "Build unit tests" ON)
option(SW_BUILD_PLUGINS "Build CLAP/VST3/AU plugins" ON)

include(FetchContent)
# Pinned third-party sources (all permissive: CLAP MIT, clap-wrapper MIT, VST3 SDK MIT, doctest MIT)
FetchContent_Declare(clap GIT_REPOSITORY https://github.com/free-audio/clap.git GIT_TAG 1.2.10 GIT_SHALLOW TRUE)
FetchContent_Declare(clap-wrapper GIT_REPOSITORY https://github.com/free-audio/clap-wrapper.git GIT_TAG v0.16.0 GIT_SHALLOW TRUE)
FetchContent_Declare(doctest URL https://raw.githubusercontent.com/doctest/doctest/v2.4.11/doctest/doctest.h DOWNLOAD_NO_EXTRACT TRUE)

# ---- DSP core (framework independent)
add_library(sw-core INTERFACE)
target_include_directories(sw-core INTERFACE core/include)

foreach(code IN LISTS SW_PRODUCTS)
    add_library(sw-${code} STATIC products/${code}/${code}.cpp)
    target_include_directories(sw-${code} PUBLIC products)

# instruments in progress: the DSP only, no plug-in yet (IN07, added 2026-10-08)
set(SW_ENGINES in07)
foreach(code IN LISTS SW_ENGINES)
    add_library(sw-${code} STATIC products/${code}/${code}.cpp)
    target_include_directories(sw-${code} PUBLIC products)

# products that are built from other products' cores
target_link_libraries(sw-vo07 PUBLIC sw-dy05 sw-dy02 sw-rv02 sw-dl01)

target_link_libraries(sw-lv09 PUBLIC sw-rs03)

# ---- the plug-in screen: the runtime and the product data become one header (build/gen/gui_assets.hpp)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(SW_GUI_ASSETS ${CMAKE_BINARY_DIR}/gen/gui_assets.hpp)
add_custom_command(OUTPUT ${SW_GUI_ASSETS}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/embed_ui.py ${SW_GUI_ASSETS}
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/embed_ui.py ${CMAKE_SOURCE_DIR}/ui/sw-ui.css ${CMAKE_SOURCE_DIR}/ui/sw-ui.js ${CMAKE_SOURCE_DIR}/ui/displays.js ${CMAKE_SOURCE_DIR}/ui/host.js ${CMAKE_SOURCE_DIR}/ui/products.json
    COMMENT "Embedding the plug-in screen")
add_custom_target(sw-gui-assets DEPENDS ${SW_GUI_ASSETS})

# ---- the product designs (docs/design/canvas) as one header per plug-in (build/gen/skins/skin_<code>.hpp); needs beautifulsoup4
set(SW_PY ${Python3_EXECUTABLE})
execute_process(COMMAND ${SW_PY} -c "import bs4" RESULT_VARIABLE SW_HAVE_BS4 OUTPUT_QUIET ERROR_QUIET)
if (NOT SW_HAVE_BS4 EQUAL 0)    # a private environment: system pythons (Homebrew) refuse pip installs
    message(STATUS "Installing beautifulsoup4 into ${CMAKE_BINARY_DIR}/pyenv (reads the design files)")
    execute_process(COMMAND ${Python3_EXECUTABLE} -m venv ${CMAKE_BINARY_DIR}/pyenv RESULT_VARIABLE SW_VENV)
    if (WIN32)
        set(SW_PY ${CMAKE_BINARY_DIR}/pyenv/Scripts/python.exe)
    else()
        set(SW_PY ${CMAKE_BINARY_DIR}/pyenv/bin/python)
    endif()
    execute_process(COMMAND ${SW_PY} -m pip install beautifulsoup4 RESULT_VARIABLE SW_PIP)
    if (NOT SW_VENV EQUAL 0 OR NOT SW_PIP EQUAL 0)
        message(FATAL_ERROR "python3 needs the beautifulsoup4 package: pip install beautifulsoup4")
    endif()
endif()
set(SW_SKIN_DIR ${CMAKE_BINARY_DIR}/gen/skins)
set(SW_SKIN_OUT "")
foreach(code IN LISTS SW_PRODUCTS)
    list(APPEND SW_SKIN_OUT ${SW_SKIN_DIR}/skin_${code}.hpp)
endforeach()
file(GLOB SW_CANVAS_FILES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/docs/design/canvas/project/*.dc.html)
add_custom_command(OUTPUT ${SW_SKIN_OUT}
    COMMAND ${SW_PY} ${CMAKE_SOURCE_DIR}/tools/gen_skins.py --emit ${SW_SKIN_DIR}
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/gen_skins.py ${CMAKE_SOURCE_DIR}/ui/specs.json ${SW_CANVAS_FILES}
    COMMENT "Generating the product screens from the design canvas")
add_custom_target(sw-gui-skins DEPENDS ${SW_SKIN_OUT})

# ---- the plug-in window's platform view: WKWebView (macOS), WebView2 (Windows), none on Linux
add_library(sw-gui-view STATIC)
target_include_directories(sw-gui-view PUBLIC plugin/clap)
if (APPLE)
    enable_language(OBJCXX)
    target_sources(sw-gui-view PRIVATE plugin/clap/gui_mac.mm)
    set_source_files_properties(plugin/clap/gui_mac.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_link_libraries(sw-gui-view PUBLIC "-framework Cocoa" "-framework WebKit")
elseif (WIN32)
    # the WebView2 SDK (headers and the static loader) comes as a NuGet package, which is a zip
    FetchContent_Declare(webview2 URL https://www.nuget.org/api/v2/package/Microsoft.Web.WebView2/1.0.2420.47 DOWNLOAD_NAME webview2.zip)
    FetchContent_MakeAvailable(webview2)
    target_sources(sw-gui-view PRIVATE plugin/clap/gui_win.cpp)
    target_include_directories(sw-gui-view PRIVATE ${webview2_SOURCE_DIR}/build/native/include)
    target_link_libraries(sw-gui-view PUBLIC ${webview2_SOURCE_DIR}/build/native/x64/WebView2LoaderStatic.lib user32 ole32 advapi32 shell32)
else()
    target_sources(sw-gui-view PRIVATE plugin/clap/gui_none.cpp)
endif()

if (SW_BUILD_PLUGINS)
    FetchContent_MakeAvailable(clap)
    set(CLAP_SDK_ROOT ${clap_SOURCE_DIR} CACHE PATH "" FORCE)
    set(CLAP_WRAPPER_DOWNLOAD_DEPENDENCIES TRUE CACHE BOOL "" FORCE)  # VST3 SDK / AudioUnit SDK
    FetchContent_MakeAvailable(clap-wrapper)

    # sw_add_plugin(<code> <display name> <AU subtype 4cc> [aufx|aumf])
    function(sw_add_plugin code name ausub)
        set(SW_CODE ${code})
        configure_file(${CMAKE_SOURCE_DIR}/plugin/clap/entry.cpp.in ${CMAKE_BINARY_DIR}/entry/${code}_entry.cpp @ONLY)
        add_library(sw-${code}-impl STATIC plugin/clap/${code}_clap.cpp)
        target_link_libraries(sw-${code}-impl PUBLIC sw-${code} clap sw-gui-view)
        target_include_directories(sw-${code}-impl PUBLIC ${CMAKE_BINARY_DIR}/gen)
        add_dependencies(sw-${code}-impl sw-gui-assets sw-gui-skins)
        target_include_directories(sw-${code}-impl PRIVATE ${SW_SKIN_DIR})
        target_compile_definitions(sw-${code}-impl PRIVATE SW_SKIN_HEADER="skin_${code}.hpp")
        set(formats CLAP VST3)
        if (APPLE)
            list(APPEND formats AUV2)
        endif()
        make_clapfirst_plugins(
            TARGET_NAME sw-${code}-plugin
            IMPL_TARGET sw-${code}-impl
            OUTPUT_NAME "${name}"
            ENTRY_SOURCE ${CMAKE_BINARY_DIR}/entry/${code}_entry.cpp
            BUNDLE_IDENTIFIER "com.seventh-well.sw-audio.${code}"
            BUNDLE_VERSION ${PROJECT_VERSION}
            COPY_AFTER_BUILD FALSE
            WINDOWS_FOLDER_VST3 TRUE  # bundle folder: required by the Steinberg validator on Windows
            PLUGIN_FORMATS ${formats}
            ASSET_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/plugins
            AUV2_MANUFACTURER_NAME "SEVENTHWELL"
            AUV2_MANUFACTURER_CODE "SvWl"
            AUV2_SUBTYPE_CODE "${ausub}"
            AUV2_INSTRUMENT_TYPE "aufx"
        )
    endfunction()
